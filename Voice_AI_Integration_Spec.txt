### Groq Integration Code

```javascript
// Using Groq (OpenAI-compatible API)
const GROQ_API_KEY = 'your-groq-api-key';
const GROQ_API_URL = 'https://api.groq.com/openai/v1/chat/completions';

async function getAIResponse(userMessage, userContext, conversationHistory) {
  const systemPrompt = buildSystemPrompt(userContext);
  
  const response = await fetch(GROQ_API_URL, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${GROQ_API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      model: 'llama-3.1-70b-versatile',  // Best free model
      messages: [
        { role: 'system', content: systemPrompt },
        ...conversationHistory,
        { role: 'user', content: userMessage }
      ],
      max_tokens: 300,      // Keep responses concise for voice
      temperature: 0.7,     // Natural variation
      stream: false         // Set true for streaming
    })
  });
  
  const data = await response.json();
  return data.choices[0].message.content;
}
```

## 🎤 WEB SPEECH API - 100% FREE & UNLIMITED

### Speech-to-Text (STT) - Unlimited

```javascript
class VoiceRecognition {
  constructor() {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    this.recognition = new SpeechRecognition();
    
    // Configuration
    this.recognition.continuous = false;      // Stop after one phrase
    this.recognition.interimResults = true;   // Show results as speaking
    this.recognition.lang = 'en-US';          // Language (supports many accents)
    this.recognition.maxAlternatives = 1;
    
    this.onResult = null;
    this.onError = null;
    
    this.recognition.onresult = (event) => {
      const result = event.results[event.results.length - 1];
      const transcript = result[0].transcript;
      const isFinal = result.isFinal;
      
      if (isFinal && this.onResult) {
        this.onResult(transcript);
      }
    };
    
    this.recognition.onerror = (event) => {
      if (this.onError) this.onError(event.error);
    };
  }
  
  start() {
    this.recognition.start();
  }
  
  stop() {
    this.recognition.stop();
  }
}

// Usage
const voiceRecognition = new VoiceRecognition();
voiceRecognition.onResult = (text) => {
  console.log('User said:', text);
  processUserInput(text);
};
voiceRecognition.start();
```

### Text-to-Speech (TTS) - Unlimited

```javascript
class TextToSpeech {
  constructor() {
    this.synth = window.speechSynthesis;
    this.voice = null;
    this.loadVoices();
  }
  
  loadVoices() {
    // Voices load asynchronously
    const loadVoiceList = () => {
      const voices = this.synth.getVoices();
      
      // Prefer natural-sounding voices
      const preferredVoices = [
        'Google UK English Female',    // Best quality in Chrome
        'Google UK English Male',
        'Google US English',
        'Microsoft Zira',              // Windows
        'Microsoft David',             // Windows
        'Samantha',                    // macOS
        'Karen',                       // macOS
        'Daniel',                      // macOS UK
      ];
      
      for (const preferred of preferredVoices) {
        const found = voices.find(v => v.name.includes(preferred));
        if (found) {
          this.voice = found;
          console.log('Selected voice:', found.name);
          break;
        }
      }
      
      // Fallback to first English voice
      if (!this.voice) {
        this.voice = voices.find(v => v.lang.startsWith('en')) || voices[0];
      }
    };
    
    // Chrome loads voices async
    if (speechSynthesis.onvoiceschanged !== undefined) {
      speechSynthesis.onvoiceschanged = loadVoiceList;
    }
    loadVoiceList();
  }
  
  speak(text) {
    return new Promise((resolve, reject) => {
      // Cancel any ongoing speech
      this.synth.cancel();
      
      const utterance = new SpeechSynthesisUtterance(text);
      
      if (this.voice) {
        utterance.voice = this.voice;
      }
      
      utterance.rate = 1.0;    // Speed (0.1 to 10)
      utterance.pitch = 1.0;   // Pitch (0 to 2)
      utterance.volume = 1.0;  // Volume (0 to 1)
      
      utterance.onend = () => resolve();
      utterance.onerror = (e) => reject(e);
      
      this.synth.speak(utterance);
    });
  }
  
  stop() {
    this.synth.cancel();
  }
  
  isSpeaking() {
    return this.synth.speaking;
  }
}

// Usage
const tts = new TextToSpeech();
await tts.speak("Hey! How can I help you with your learning today?");
```

### Tips for Better Web Speech API Voices

1. **Use Chrome** - Has the best Google voices
2. **Use UK English voices** - Sound more natural than US
3. **Adjust rate to 0.95-1.0** - Slightly slower sounds more natural
4. **Add SSML pauses** - Insert commas and periods for natural breathing

```javascript
// Add natural pauses
function addNaturalPauses(text) {
  return text
    .replace(/\. /g, '. ... ')      // Pause after sentences
    .replace(/! /g, '! ... ')       // Pause after exclamations
    .replace(/, /g, ', ');          // Brief pause after commas
}
```

### Smart Fallback Strategy

```javascript
class AIProvider {
  constructor() {
    this.providers = [
      { name: 'groq', fn: this.callGroq.bind(this) },
      { name: 'gemini', fn: this.callGemini.bind(this) },
      { name: 'openrouter', fn: this.callOpenRouter.bind(this) }
    ];
  }
  
  async getResponse(message, context, history) {
    for (const provider of this.providers) {
      try {
        console.log(`Trying ${provider.name}...`);
        return await provider.fn(message, context, history);
      } catch (error) {
        console.warn(`${provider.name} failed:`, error.message);
        continue;  // Try next provider
      }
    }
    throw new Error('All AI providers failed');
  }
}
```

## 🔗 MAIN CLASS

```javascript
class InternPathVoiceAssistant {
  constructor(config) {
    this.groqApiKey = config.groqApiKey;
    this.authToken = config.authToken;
    this.apiBaseUrl = config.apiBaseUrl;
    
    this.conversationHistory = [];
    this.isListening = false;
    this.isSpeaking = false;
    
    // Initialize components
    this.stt = new VoiceRecognition();
    this.tts = new TextToSpeech();
    
    // Set up STT callback
    this.stt.onResult = (transcript) => {
      this.processUserInput(transcript);
    };
  }
  
  async startListening() {
    if (this.isSpeaking) {
      this.tts.stop();  // Interrupt if speaking
    }
    this.isListening = true;
    this.stt.start();
    this.onStateChange?.('listening');
  }
  
  stopListening() {
    this.isListening = false;
    this.stt.stop();
    this.onStateChange?.('idle');
  }
  
  async processUserInput(transcript) {
    this.onStateChange?.('processing');
    this.onTranscript?.(transcript);
    
    try {
      // 1. Add to history
      this.conversationHistory.push({ role: 'user', content: transcript });
      
      // 2. Fetch user context
      const context = await this.fetchUserContext();
      
      // 3. Get AI response
      const response = await this.callGroq(transcript, context);
      
      // 4. Add to history
      this.conversationHistory.push({ role: 'assistant', content: response });
      
      // 5. Speak response
      this.onResponse?.(response);
      this.onStateChange?.('speaking');
      this.isSpeaking = true;
      
      await this.tts.speak(response);
      
      this.isSpeaking = false;
      this.onStateChange?.('idle');
      
    } catch (error) {
      this.onError?.(error);
      this.onStateChange?.('error');
    }
  }
  
  async callGroq(message, context) {
    const response = await fetch('https://api.groq.com/openai/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.groqApiKey}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: 'llama-3.1-70b-versatile',
        messages: [
          { role: 'system', content: this.buildSystemPrompt(context) },
          ...this.conversationHistory.slice(-10)  // Keep last 10 messages
        ],
        max_tokens: 300,
        temperature: 0.7
      })
    });
    
    const data = await response.json();
    return data.choices[0].message.content;
  }
  
  async fetchUserContext() {
    const headers = { 'Authorization': `Bearer ${this.authToken}` };
    
    const [profileRes, skillsRes, streakRes] = await Promise.all([
      fetch(`${this.apiBaseUrl}/users/profile`, { headers }),
      fetch(`${this.apiBaseUrl}/skills`, { headers }),
      fetch(`${this.apiBaseUrl}/progress/streak`, { headers })
    ]);
    
    const profile = await profileRes.json();
    const skills = await skillsRes.json();
    const streak = await streakRes.json();
    
    return {
      name: profile.data?.name || 'there',
      goal: profile.data?.goal || 'tech',
      streak: streak.data?.current_streak || 0,
      completedSkills: skills.data?.filter(s => s.is_completed).length || 0,
      totalSkills: skills.data?.length || 0
    };
  }
  
  buildSystemPrompt(context) {
    return `You are the InternPath Voice Assistant - a friendly AI mentor.

Current User:
- Name: ${context.name}
- Goal: ${context.goal} Developer
- Current Streak: ${context.streak} days
- Completed: ${context.completedSkills}/${context.totalSkills} skills

Speak naturally like a friend. Keep responses to 2-4 sentences. Use their name. Be encouraging!`;
  }
}
```
